11/// Converts a UTF-16 code unit (C# `char`) into `(raw number, symbol marker)`.
12#[derive(Clone, Copy, Debug)]
13pub struct CharToUnicodeSymbolConverter<T> {
14    address_to_number: AddressToRawNumberConverter,
15    symbol_marker: T,
16}
17
18impl<T: LinkReference> CharToUnicodeSymbolConverter<T> {
19    /// Creates the converter with a raw-number encoder and caller-owned marker.
20    #[must_use]
21    pub const fn new(address_to_number: AddressToRawNumberConverter, symbol_marker: T) -> Self {
22        Self {
23            address_to_number,
24            symbol_marker,
25        }
26    }
27
28    fn raw_number<L: Doublets<T>>(&self, links: &L, code_unit: u16) -> Result<T, SequenceError<T>> {
29        let address =
30            T::try_from(code_unit).map_err(|_| SequenceError::CodeUnitDoesNotFit(code_unit))?;
31        let raw = self.address_to_number.convert(address)?;
32        if !links.constants().is_external(raw) {
33            return Err(SequenceError::ExternalReferencesDisabled);
34        }
35        Ok(raw)
36    }
37
38    /// Stores or reuses a symbol, preserving every code unit including lone surrogates.
39    /// Rejects code units outside the address range or stores without external support.
40    pub fn convert<L: Doublets<T>>(
41        &self,
42        links: &mut L,
43        code_unit: u16,
44    ) -> Result<T, SequenceError<T>> {
45        let raw = self.raw_number(links, code_unit)?;
46        Ok(links.get_or_create(raw, self.symbol_marker)?)
47    }
48}

50/// Converts a marked Unicode symbol back into a UTF-16 code unit.
51#[derive(Clone, Copy, Debug)]
52pub struct UnicodeSymbolToCharConverter<T, M = TargetMatcher<T>> {
53    number_to_address: RawNumberToAddressConverter,
54    matcher: M,
55    address: std::marker::PhantomData<T>,
56}
57
58impl<T: LinkReference, M: CriterionMatcher<T>> UnicodeSymbolToCharConverter<T, M> {
59    /// Creates a decoder with a raw-number decoder and symbol criterion.
60    #[must_use]
61    pub const fn new(number_to_address: RawNumberToAddressConverter, matcher: M) -> Self {
62        Self {
63            number_to_address,
64            matcher,
65            address: std::marker::PhantomData,
66        }
67    }
68
69    /// Checks the symbol criterion and decodes its source, rejecting oversized payloads.
70    pub fn convert<L: Doublets<T> + ?Sized>(
71        &self,
72        links: &L,
73        symbol: T,
74    ) -> Result<u16, SequenceError<T>> {
75        let link = links.try_get_link(symbol)?;
76        if !self.matcher.is_matched(links, symbol) {
77            return Err(SequenceError::NotUnicodeSymbol(symbol));
78        }
79        let unit = self.number_to_address.convert(link.source);
80        unit.try_into()
81            .map_err(|_| SequenceError::CodeUnitOutOfRange(unit))
82    }
83}

85/// Encodes strings as UTF-16 symbols in a marked balanced sequence.
86#[derive(Clone, Copy, Debug)]
87pub struct StringToUnicodeSequenceConverter<T> {
88    symbol_converter: CharToUnicodeSymbolConverter<T>,
89    balanced_converter: BalancedVariantConverter,
90    sequence_marker: T,
91}
92
93impl<T: LinkReference> StringToUnicodeSequenceConverter<T> {
94    /// Creates a converter from the symbol encoder, balancing strategy and marker.
95    #[must_use]
96    pub const fn new(
97        symbol_converter: CharToUnicodeSymbolConverter<T>,
98        balanced_converter: BalancedVariantConverter,
99        sequence_marker: T,
100    ) -> Self {
101        Self {
102            symbol_converter,
103            balanced_converter,
104            sequence_marker,
105        }
106    }
107
108    /// Encodes a Rust string using UTF-16, including surrogate pairs for non-BMP text.
109    /// An empty string returns the sequence marker without creating any links.
110    pub fn convert<L: Doublets<T>>(
111        &self,
112        links: &mut L,
113        content: &str,
114    ) -> Result<T, SequenceError<T>> {
115        self.convert_utf16(links, &content.encode_utf16().collect::<Vec<_>>())
116    }
117
118    /// Encodes raw UTF-16 units, preserving unpaired surrogates from C# strings.
119    /// Validates numeric/external ranges before creating symbols; store failures
120    /// can still leave already-created links, as store writes are not transactional.
121    pub fn convert_utf16<L: Doublets<T>>(
122        &self,
123        links: &mut L,
124        units: &[u16],
125    ) -> Result<T, SequenceError<T>> {
126        if units.is_empty() {
127            return Ok(self.sequence_marker);
128        }
129        let raw_numbers = units
130            .iter()
131            .map(|&unit| self.symbol_converter.raw_number(links, unit))
132            .collect::<Result<Vec<_>, _>>()?;
133        let symbols = raw_numbers
134            .into_iter()
135            .map(|raw| links.get_or_create(raw, self.symbol_converter.symbol_marker))
136            .collect::<Result<Vec<_>, _>>()?;
137        let sequence = self.balanced_converter.convert(links, &symbols)?;
138        Ok(links.get_or_create(sequence, self.sequence_marker)?)
139    }
140}

142/// Decodes a marked Unicode sequence into UTF-16 units or a Rust string.
143#[derive(Clone, Copy, Debug)]
144pub struct UnicodeSequenceToStringConverter<
145    T,
146    S = TargetMatcher<T>,
147    E = TargetMatcher<T>,
148    M = TargetMatcher<T>,
149> {
150    sequence_matcher: S,
151    walker: RightSequenceWalker<E>,
152    symbol_converter: UnicodeSymbolToCharConverter<T, M>,
153    sequence_marker: T,
154}
155
156impl<T, S, E, M> UnicodeSequenceToStringConverter<T, S, E, M>
157where
158    T: LinkReference,
159    S: CriterionMatcher<T>,
160    E: CriterionMatcher<T>,
161    M: CriterionMatcher<T>,
162{
163    /// Creates a decoder with sequence/symbol criteria, a walker and empty marker.
164    #[must_use]
165    pub const fn new(
166        sequence_matcher: S,
167        walker: RightSequenceWalker<E>,
168        symbol_converter: UnicodeSymbolToCharConverter<T, M>,
169        sequence_marker: T,
170    ) -> Self {
171        Self {
172            sequence_matcher,
173            walker,
174            symbol_converter,
175            sequence_marker,
176        }
177    }
178
179    /// Decodes a sequence, returning an error for unpaired UTF-16 surrogates.
180    pub fn convert<L: Doublets<T> + ?Sized>(
181        &self,
182        links: &L,
183        sequence: T,
184    ) -> Result<String, SequenceError<T>> {
185        Ok(String::from_utf16(&self.convert_utf16(links, sequence)?)?)
186    }
187
188    /// Decodes the exact stored UTF-16 units, including unpaired surrogates.
189    /// The configured empty marker returns no units without a store lookup.
190    pub fn convert_utf16<L: Doublets<T> + ?Sized>(
191        &self,
192        links: &L,
193        sequence: T,
194    ) -> Result<Vec<u16>, SequenceError<T>> {
195        if sequence == self.sequence_marker {
196            return Ok(Vec::new());
197        }
198        let link = links.try_get_link(sequence)?;
199        if !self.sequence_matcher.is_matched(links, sequence) {
200            return Err(SequenceError::NotUnicodeSequence(sequence));
201        }
202        if link.source == links.constants().null {
203            return Err(crate::Error::NotExists(link.source).into());
204        }
205        self.walker
206            .iter(links, link.source)
207            .map(|symbol| self.symbol_converter.convert(links, symbol?))
208            .collect()
209    }
