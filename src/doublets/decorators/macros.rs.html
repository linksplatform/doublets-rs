3/// Emits the struct, constructors and the [`Doublets`](crate::Doublets) forwarding
4/// impl shared by every decorator that wraps a single store.
5macro_rules! decorator_struct {
6    (
7        $(#[$attr:meta])*
8        $name:ident
9    ) => {
10        $(#[$attr])*
11        pub struct $name<T: LinkReference, L: Doublets<T>> {
12            links: L,
13            _marker: PhantomData<fn(T)>,
14        }
15
16        impl<T: LinkReference, L: Doublets<T>> $name<T, L> {
17            #[doc = concat!("Wraps `links` in a [`", stringify!($name), "`].")]
18            #[inline]
19            pub const fn new(links: L) -> Self {
20                Self {
21                    links,
22                    _marker: PhantomData,
23                }
24            }
25
26            /// Returns a shared reference to the wrapped store.
27            #[inline]
28            #[must_use]
29            pub const fn inner(&self) -> &L {
30                &self.links
31            }
32
33            /// Returns a mutable reference to the wrapped store.
34            #[inline]
35            #[must_use]
36            pub const fn inner_mut(&mut self) -> &mut L {
37                &mut self.links
38            }
39
40            /// Unwraps the decorator and returns the wrapped store.
41            #[inline]
42            #[must_use]
43            pub fn into_inner(self) -> L {
44                self.links
45            }
46        }
47
48        impl<T: LinkReference, L: Doublets<T> + Debug> Debug for $name<T, L> {
49            fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
50                f.debug_tuple(stringify!($name)).field(&self.links).finish()
51            }
52        }
53
54        impl<T: LinkReference, L: Doublets<T>> Doublets<T> for $name<T, L> {
55            #[inline]
56            fn get_link(&self, index: T) -> Option<Link<T>> {
57                self.links.get_link(index)
58            }
59        }
60    };
61}

63/// Emits verbatim [`Links`](crate::Links) forwarding methods for every operation a
64/// decorator does not intercept.
65///
66/// [`Links::constants`](crate::Links::constants) is always forwarded, because no
67/// decorator changes the store's constants.
68macro_rules! forward {
69    ($($method:ident),* $(,)?) => {
70        #[inline]
71        fn constants(&self) -> &LinksConstants<T> {
72            self.links.constants()
73        }
74
75        $(forward!(@method $method);)*
76    };
77    (@method count_links) => {
78        #[inline]
79        fn count_links(&self, query: &[T]) -> T {
80            self.links.count_links(query)
81        }
82    };
83    (@method each_links) => {
84        #[inline]
85        fn each_links(&self, query: &[T], handler: ReadHandler<'_, T>) -> Flow {
86            self.links.each_links(query, handler)
87        }
88    };
89    (@method create_links) => {
90        #[inline]
91        fn create_links(
92            &mut self,
93            query: &[T],
94            handler: WriteHandler<'_, T>,
95        ) -> Result<Flow, Error<T>> {
96            self.links.create_links(query, handler)
97        }
98    };
99    (@method update_links) => {
100        #[inline]
101        fn update_links(
102            &mut self,
103            query: &[T],
104            change: &[T],
105            handler: WriteHandler<'_, T>,
106        ) -> Result<Flow, Error<T>> {
107            self.links.update_links(query, change, handler)
108        }
109    };
110    (@method delete_links) => {
111        #[inline]
112        fn delete_links(
113            &mut self,
114            query: &[T],
115            handler: WriteHandler<'_, T>,
116        ) -> Result<Flow, Error<T>> {
117            self.links.delete_links(query, handler)
118        }
119    };
120}

122pub(crate) use {decorator_struct, forward};
