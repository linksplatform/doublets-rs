9/// The mutation represented by a [`Transition`].
10#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
11pub enum TransitionKind {
12    /// A previously absent link was created.
13    Create,
14    /// The contents of an existing link changed.
15    Update,
16    /// An existing link was deleted.
17    Delete,
18}
19
20/// A reversible mutation, including both link states.
21#[derive(Debug, Clone, PartialEq, Eq)]
22pub struct Transition<T: LinkReference> {
23    /// Journal-local transaction identifier.
24    pub transaction_id: u64,
25    /// Journal-local, monotonically increasing write sequence.
26    pub sequence: u64,
27    /// The kind of mutation.
28    pub kind: TransitionKind,
29    /// Link state before mutation; the null link for creation.
30    pub before: Link<T>,
31    /// Link state after mutation; the null link for deletion.
32    pub after: Link<T>,
33}
34
35impl<T: LinkReference> Transition<T> {
36    /// Constructs a transition, inferring its kind from the null side.
37    pub fn new(transaction_id: u64, sequence: u64, before: Link<T>, after: Link<T>) -> Self {
38        let kind = if before.is_null() {
39            TransitionKind::Create
40        } else if after.is_null() {
41            TransitionKind::Delete
42        } else {
43            TransitionKind::Update
44        };
45        Self {
46            transaction_id,
47            sequence,
48            kind,
49            before,
50            after,
51        }
52    }
53
54    /// Applies this write without generating another journal record.
55    ///
56    /// Repeated application is harmless. The store must support raw writes and
57    /// exact-address restoration through sequential allocation (as both supplied
58    /// stores do). This reconstructs the graph with O(number of links + highest
59    /// address) store operations, plus sorting and backend index costs.
60    pub fn apply<L: Doublets<T>>(&self, links: &mut L) -> Result<(), Error<T>> {
61        self.validate(links)?;
62        Self::set_state(links, &self.before, &self.after)
63    }
64
65    /// Reverts this write without generating another journal record.
66    ///
67    /// Like [`Self::apply`], this is idempotent and preserves link addresses.
68    pub fn revert<L: Doublets<T>>(&self, links: &mut L) -> Result<(), Error<T>> {
69        self.validate(links)?;
70        Self::set_state(links, &self.after, &self.before)
71    }
72
73    pub(super) fn validate<L: Doublets<T>>(&self, links: &L) -> Result<(), Error<T>> {
74        let inferred = Self::new(
75            self.transaction_id,
76            self.sequence,
77            self.before.clone(),
78            self.after.clone(),
79        )
80        .kind;
81        if self.kind != inferred
82            || (self.before.is_null() && self.after.is_null())
83            || (!self.before.is_null()
84                && !self.after.is_null()
85                && self.before.index != self.after.index)
86            || [&self.before, &self.after]
87                .iter()
88                .any(|link| !link.is_null() && !links.constants().is_internal(link.index))
89        {
90            return Err(std::io::Error::new(
91                std::io::ErrorKind::InvalidData,
92                "invalid transition states",
93            )
94            .into());
95        }
96        Ok(())
97    }
98
99    fn set_state<L: Doublets<T>>(
100        links: &mut L,
101        from: &Link<T>,
102        to: &Link<T>,
103    ) -> Result<(), Error<T>> {
104        let index = if to.is_null() { from.index } else { to.index };
105        let mut snapshot = super::snapshot(links);
106        snapshot.retain(|link| link.index != index);
107        if !to.is_null() {
108            snapshot.push(to.clone());
109        }
110        restore(links, &snapshot)
111    }
112}

114/// A durable recovery record.
115///
116/// Start and commit snapshots bridge the gap between a store mutation and its
117/// after-write callback, and remove the need for a store-specific flush hook.
118#[derive(Debug, Clone, PartialEq, Eq)]
119pub enum JournalEntry<T: LinkReference> {
120    /// Persisted before the transaction can modify the store.
121    Begin {
122        /// Journal-local transaction identifier.
123        transaction_id: u64,
124        /// Complete graph before any writes.
125        snapshot: Vec<Link<T>>,
126    },
127    /// One callback's before/after pair, in write order.
128    Transition(Transition<T>),
129    /// Persisted before commit returns successfully.
130    Commit {
131        /// The transaction being committed.
132        transaction_id: u64,
133        /// Complete committed graph, used for redo.
134        snapshot: Vec<Link<T>>,
135    },
136    /// Persisted after restoring the begin snapshot.
137    Rollback {
138        /// The transaction being rolled back.
139        transaction_id: u64,
140    },
141}
