3use data::{Flow, LinkReference};
4
5use crate::{data::WriteHandler, Doublets, Error};

7/// Releases already reset links, independently of the endpoint reset policy.
8///
9/// Implemented by both memory stores. Unlike their compatibility `delete*`
10/// helpers, these operations never update endpoints. To compose safe release
11/// with decorators, use [`CheckedRelease`](crate::decorators::CheckedRelease).
12///
13/// Incoming references are allowed and retain their values. The split layout
14/// moves their index entries to external trees; the unit layout uses global
15/// endpoint trees. Reuse of the address makes these references internal again.
16/// Rejecting or deleting usages is a separate decorator policy.
17///
18/// # Examples
19///
20/// ```
21/// use doublets::{mem::Release, unit, Doublets};
22/// use mem::Global;
23/// # fn main() -> Result<(), doublets::Error<u32>> {
24/// let mut store = unit::Store::<u32, _>::new(Global::new())?;
25/// let index = store.create_point()?;
26/// store.update(index, 0, 0)?;
27/// assert_eq!(store.release(index)?, index);
28/// assert!(!store.exist(index));
29/// # Ok(())
30/// # }
31/// ```
32pub trait Release<T: LinkReference>: Doublets<T> {
33    /// Releases a slot without checking the contract or resetting its contents.
34    ///
35    /// Performs only structural bookkeeping (including incoming-usage index
36    /// reclassification where required) and free-list/tail maintenance. Calls
37    /// `handler` once with `(index, 0, 0)` and [`crate::Link::nothing()`], after
38    /// release, returning its flow. `Break` does not undo the release.
39    ///
40    /// # Safety
41    ///
42    /// - `index` must be a live allocated address, never null, a service constant,
43    ///   an unused slot or an address that has already been released.
44    /// - Both endpoints must be zero and their outgoing index memberships must
45    ///   already be detached. Creating an empty link or updating a live link to
46    ///   `(0, 0)` through the store API establishes this condition. Merely
47    ///   overwriting raw endpoint memory does not.
48    /// - All incoming source/target index entries must be consistent with the
49    ///   live address. They need not be absent: release preserves them, including
50    ///   links using the address in both endpoints. Do not manipulate raw index
51    ///   memory to bypass that invariant.
52    ///
53    /// The caller must uphold all conditions even in release builds. Safe code
54    /// should use [`Release::release_with`] or [`Release::release`].
55    unsafe fn release_unchecked(&mut self, index: T, handler: WriteHandler<'_, T>) -> Flow;
56
57    /// Checks that `index` is live and reset, then releases it without an update.
58    ///
59    /// Returns [`Error::NotExists`] for a non-live address, or
60    /// [`Error::NonNullContents`] when either endpoint is nonzero. Errors leave
61    /// the store unchanged and do not call the handler. Incoming usages remain.
62    #[inline]
63    fn release_with(&mut self, index: T, handler: WriteHandler<'_, T>) -> Result<Flow, Error<T>> {
64        let link = self.try_get_link(index)?;
65        if link.source != T::from_byte(0) || link.target != T::from_byte(0) {
66            return Err(Error::NonNullContents(link));
67        }
68        // SAFETY: The address is live and reset. Safe store operations maintain
69        // outgoing memberships and incoming index consistency.
70        Ok(unsafe { self.release_unchecked(index, handler) })
71    }
72
73    /// Checks and releases an already reset live link, returning its former index.
74    #[inline]
75    fn release(&mut self, index: T) -> Result<T, Error<T>> {
76        self.release_with(index, &mut |_, _| Flow::Continue)
77            .map(|_| index)
78    }
79}
